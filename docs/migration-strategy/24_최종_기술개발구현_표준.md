### 4.1 CREATOR Engine 구현

```python
# LangGraph 기반 CREATOR 7단계 엔진
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver

class CreatorState(TypedDict):
    document: str           # 원고 텍스트
    concept: dict           # C: 컨셉 분석 결과
    route: dict             # R: 경로 설계 결과
    experience: dict        # E: 경험 설계 결과
    artifacts: list         # A: 생성된 산출물
    transform: dict         # T: 변화 적용 결과
    operation: dict         # O: 최적 운영 결과
    reflection: dict        # R: 성찰 결과
    human_feedback: str     # Human-in-the-Loop 피드백
    status: str             # 현재 단계

# 각 노드는 Gemini 2.5 Pro 호출
async def concept_node(state: CreatorState) -> CreatorState:
    """C: 컨셉 분석 - 학습 목표, 대상, 범위 도출"""
    result = await gemini.analyze(
        prompt=CONCEPT_PROMPT,
        document=state["document"]
    )
    return {"concept": result, "status": "concept_complete"}
```

### 4.3 Generator Engine 구현

```typescript
// Next.js 16+ Server Action - 영상 생성 트리거
'use server';

import { generateVideo } from '@/lib/veo';
import { generateScript } from '@/lib/gemini';

export async function generateCourseContent(formData: FormData) {
  const document = formData.get('document') as File;

  // 1. 스크립트 생성 (Gemini 2.5 Pro)
  const script = await generateScript(document);

  // 2. 4K 영상 생성 (Veo 3.1 + Cloud Run GPU)
  const video = await generateVideo({
    script,
    resolution: '4K',
    codec: 'AV1',
    model: 'veo-3.1-lite', // 비용 최적화
  });

  // 3. PPT 생성 (Gemini)
  const ppt = await generatePPT(script);

  // 4. 웹 콘텐츠 렌더링 (Claude)
  const webContent = await generateWebContent(script);

  return { video, ppt, webContent };
}
```
